// Reject AutoMail 4 ขั้น  สรุปงาน Reject ทุก Vendor ทีละขั้น คืนแถวเดียว HTML ประกอบจบใน DAX (รุ่นขั้นเดียว 3.0.4 เก็บไว้ที่ GetData_Summary_304only.dax.bak)
// พี่ปอนด์ 08/11/2026: เพิ่ม 1.1.1 , 1.1.2 , 2.0.2 นับงาน Reject จากคอลัมน์ REASON มีค่า (REJECT_REASON_FLAG) , 3.0.4 นับจาก REJECT_REASON มีค่า (REJECT_FLAG) เหมือนเดิม
//   แต่ละขั้นมีกราฟและตารางแยกปี 2025 2026 2027 , การ์ด TOTAL Reject รวม 4 ขั้น , หัวเรื่องเมลภาษาอังกฤษ , ขั้นที่ไม่มีงานขึ้นก้อนบอกว่าไม่มีงาน Reject
// พี่ปอนด์ 04/11/2026: VENDOR = Sub_New , AGING_GROUP <=7 , <=15 , <=30 , >30 Day (>30 ตัวแดง) , Vendor ที่ไม่มีงานยังขึ้นในตารางเป็นช่องว่างให้เทียบกันได้
// AGING ตอนนี้ใช้คอลัมน์คำนวณในไฟล์ (Aging_Days , Aging Group) วันที่ฐานข้อมูลมี AGING_DAY , AGING_GROUP แก้บรรทัดที่มีป้าย <-- AGING
DEFINE
  // <-- ขั้นที่นับ: รหัสขั้น (SUB_STAGE ขึ้นต้นด้วย 5 ตัวนี้) , คอลัมน์ที่ใช้ตัดสินว่า Reject , ชื่อสำรองถ้าข้อมูลไม่มีขั้นนี้เลย
  VAR Stages_ =
    DATATABLE("@s", STRING, "@src", STRING, "@name", STRING, "@small", BOOLEAN, "@chart6", BOOLEAN, {   // @small = ขั้นงานน้อย ใช้กับสวิตช์ TableOnlySmall_ , @chart6 = ขั้นที่มีกราฟในแบบ 6
      {"1.1.1", "REASON", "1.1.1 Rejected Draft Contract (From SMT)", TRUE, FALSE},
      {"1.1.2", "REASON", "1.1.2 Return Pre-Contract to Edit", TRUE, TRUE},
      {"2.0.2", "REASON", "2.0.2 Back Contract to Draft (SMT Process)", TRUE, FALSE},
      {"3.0.4", "REJECT_REASON", "3.0.4 Review Completed Contract (SMT Process)", FALSE, TRUE}})
  VAR Years_ = {2025, 2026, 2027}                   // <-- ปีที่สัญญาเดิมหมด (OLD_CONTRACT_END_DATE) ที่นับ พี่ปอนด์ 04/11/2026 เอาแค่ 2025 ถึง 2027
  VAR YearsText_ = "2025 ถึง 2027"                  // <-- ข้อความช่วงปีในหัวเมล แก้คู่กับ Years_
  VAR Hide_ = {" No Assign", ""}                    // <-- Sub_New ที่ไม่นับเป็น Vendor
  VAR TableOnlyAll_ = TRUE()                       // <-- TRUE แบบ 7: ทุกขั้นรวม 3.0.4 มีแต่ตารางรวมสามปี ไม่มีกราฟ ไม่ขึ้น Vendor ที่ไม่มีงาน (พี่ปอนด์ 09/11/2026)
  VAR SubTotalYear_ = FALSE()                      // <-- TRUE ตารางรวมมีแถว Sub Total ต่อปี เฉพาะปีที่มี Vendor ตั้งแต่ 2 ราย (พี่ปอนด์ 09/11/2026)
  VAR Layout6_ = FALSE()                            // <-- TRUE แบบ 6: ขั้น @chart6 (1.1.2 3.0.4) มีกราฟ ทุกขั้นใช้ตารางรวมสามปี ไม่สนสวิตช์ TableOnlySmall_ CombineYearsSmall_ (พี่ปอนด์ 08/11/2026)
  VAR TableOnlySmall_ = TRUE()                     // <-- TRUE ขั้น @small (1.1.1 1.1.2 2.0.2) มีแต่ตาราง ไม่มีกราฟ ขึ้นเฉพาะแถวที่มีงาน , 3.0.4 เหมือนเดิม (พี่ปอนด์ 08/11/2026 แบบ 3)
  VAR ShowEmptyYearSmall_ = TRUE()                // <-- TRUE ใช้คู่ CombineYearsSmall_ ปีที่ไม่มีงานยังขึ้นแถว ไม่มีงาน Reject (พี่ปอนด์ 08/11/2026 แบบ 5)
  VAR CombineYearsSmall_ = TRUE()                  // <-- TRUE ใช้คู่ TableOnlySmall_ ขั้น @small รวมสามปีเป็นตารางเดียวมีคอลัมน์ ปี (พี่ปอนด์ 08/11/2026 แบบ 4)
  VAR ShowEmpty_ = TRUE()                          // <-- ตารางขึ้น Vendor ที่ว่างในปีนั้นด้วยไหม TRUE แบบเดิม , FALSE ซ่อนแถวว่าง เมลสั้นลง (พี่ปอนด์ 08/11/2026 ขอดู 2 แบบ)
  VAR Bars_ = 170                                  // ความยาวแท่งยาวสุด (px) กราฟสามปีวางแถวเดียวกัน
  VAR StageKeys_ = SELECTCOLUMNS(Stages_, "@k", [@s])
  VAR ReasonKeys_ = SELECTCOLUMNS(FILTER(Stages_, [@src] = "REASON"), "@k", [@s])
  VAR StageList_ = CONCATENATEX(Stages_, [@s], ", ", [@s], ASC)
  VAR Vendors_ = FILTER(DISTINCT('RENEW CONTRACT TASK'[Sub_New]), NOT ('RENEW CONTRACT TASK'[Sub_New] IN Hide_))
  // งาน Reject ของทุกขั้น ตาราง Pick_ ใช้ร่วมกับแถวไฟล์ Excel (build_dax.py) ให้เมลกับไฟล์แนบนับชุดเดียวกันเสมอ
  VAR Pick_ =
    FILTER('RENEW CONTRACT TASK',
      VAR s = LEFT('RENEW CONTRACT TASK'[SUB_STAGE], 5)
      RETURN s IN StageKeys_
        && YEAR('RENEW CONTRACT TASK'[OLD_CONTRACT_END_DATE]) IN Years_
        && IF(s IN ReasonKeys_, 'RENEW CONTRACT TASK'[REJECT_REASON_FLAG], 'RENEW CONTRACT TASK'[REJECT_FLAG]) = "Yes")
  VAR Base_ =
    SELECTCOLUMNS(Pick_,
      "S", LEFT('RENEW CONTRACT TASK'[SUB_STAGE], 5),
      "V", 'RENEW CONTRACT TASK'[Sub_New],
      "A", 'RENEW CONTRACT TASK'[Aging Group],       // <-- AGING เปลี่ยนเป็น [AGING_GROUP]
      "Y", YEAR('RENEW CONTRACT TASK'[OLD_CONTRACT_END_DATE]))
  VAR Upd_ = MAX('RENEW CONTRACT TASK'[UPDATED_DATE])
  VAR YearsT_ = SELECTCOLUMNS(Years_, "@yy", [Value])
  VAR StagesT_ = SELECTCOLUMNS(Stages_, "@ss", [@s])
  // Vendor คูณปี คูณขั้น นับทีละช่อง
  VAR RowsY_ =
    ADDCOLUMNS(
      CROSSJOIN(Vendors_, YearsT_, StagesT_),
      "@v", 'RENEW CONTRACT TASK'[Sub_New],
      "@h", SUBSTITUTE(SUBSTITUTE(SUBSTITUTE('RENEW CONTRACT TASK'[Sub_New], "&", "&amp;"), "<", "&lt;"), ">", "&gt;"),
      "@a1", COUNTROWS(FILTER(Base_, [S] = [@ss] && [V] = 'RENEW CONTRACT TASK'[Sub_New] && [Y] = [@yy] && [A] = "<=7 Day")) + 0,
      "@a2", COUNTROWS(FILTER(Base_, [S] = [@ss] && [V] = 'RENEW CONTRACT TASK'[Sub_New] && [Y] = [@yy] && [A] = "<=15 Day")) + 0,
      "@a3", COUNTROWS(FILTER(Base_, [S] = [@ss] && [V] = 'RENEW CONTRACT TASK'[Sub_New] && [Y] = [@yy] && [A] = "<=30 Day")) + 0,
      "@a4", COUNTROWS(FILTER(Base_, [S] = [@ss] && [V] = 'RENEW CONTRACT TASK'[Sub_New] && [Y] = [@yy] && [A] = ">30 Day")) + 0,
      "@n", COUNTROWS(FILTER(Base_, [S] = [@ss] && [V] = 'RENEW CONTRACT TASK'[Sub_New] && [Y] = [@yy])) + 0)
  VAR Y0_ = MINX(YearsT_, [@yy])
  VAR N_ = SUMX(RowsY_, [@n])
  VAR Ages_ = {"<=7 Day", "<=15 Day", "<=30 Day", ">30 Day"}
  VAR OddRows_ = FILTER(Base_, NOT ([A] IN Ages_) && NOT ([V] IN Hide_))
  VAR Odd_ = COUNTROWS(OddRows_) + 0
  VAR OddList_ = CONCATENATEX(DISTINCT(SELECTCOLUMNS(OddRows_, "a", COALESCE([A], "ว่าง"))), [a], ", ")
  // ชื่อขั้นเต็มกับ GROUP จากข้อมูลจริง ไม่มีในข้อมูลใช้ชื่อสำรอง
  VAR StageInfo_ =
    ADDCOLUMNS(Stages_,
      "@full", COALESCE(MAXX(FILTER('RENEW CONTRACT TASK', LEFT('RENEW CONTRACT TASK'[SUB_STAGE], 5) = [@s]), 'RENEW CONTRACT TASK'[SUB_STAGE]), [@name]),
      "@grp", COALESCE(MAXX(FILTER('RENEW CONTRACT TASK', LEFT('RENEW CONTRACT TASK'[SUB_STAGE], 5) = [@s]), 'RENEW CONTRACT TASK'[GROUP]), "-"))
  // พี่ปอนด์ 04/11/2026 สีและตัวอักษรชุดเดียวกับเมล 0 Wait Assign (หัวตาราง #D9D9D9 ตัว #222 เส้น #D9D9D9 , เนื้อตัวปกติ , ข้อความรอง #444 #666 #888)
  // Outlook ใช้ Word engine สไตล์ inline ทุกช่อง ใช้ตารางแทน div
  // ฟอนต์เดียวทั้งเมล Leelawadee UI มีทั้งไทยและอังกฤษในตัว , ขนาด: หัวเรื่อง 20 , ตัวเลขการ์ด 40 , ข้อความทั่วไปและหัวข้อ 13 , ตาราง คำอธิบายสี หมายเหตุ 12 , ตัวเลขในแท่ง 11
  VAR F_ = "font-family:Leelawadee UI,Leelawadee,Segoe UI,Tahoma,sans-serif;font-size:12px;"
  VAR TH_ = "<th nowrap style='background:#D9D9D9;border:1px solid #D9D9D9;padding:6px 8px;" & F_ & "font-weight:bold;color:#222;text-align:"
  VAR TD_ = "<td style='border:1px solid #D9D9D9;padding:5px 8px;" & F_ & "color:#222;"
  VAR TS_ = "<td nowrap style='background:#F7F7F7;border:1px solid #D9D9D9;padding:5px 8px;" & F_ & "font-weight:bold;color:#222;text-align:"   // แถว Sub Total ต่อปี (พี่ปอนด์ 09/11/2026)
  VAR TF_ = "<td nowrap style='background:#F2F2F2;border:1px solid #D9D9D9;padding:6px 8px;" & F_ & "font-weight:bold;color:#222;text-align:"
  VAR Z_ = ""   // พี่ปอนด์ 04/11/2026 ช่องที่เป็น 0 เว้นว่าง
  VAR K_ = COUNTROWS(Vendors_)
  VAR YP_ = "<p style='" & F_ & "font-size:13px;color:#222;margin:"   // หัวปีของกราฟและตาราง
  VAR NoJob_ = "<p style='margin:0;" & F_ & "color:#666'>ไม่มีงาน Reject</p>"
  // กราฟแท่งซ้อนต่อ Vendor แบ่งสีตามอายุงาน แบบเมล Call Center , ชื่อ (ยอด) , Vendor ที่เป็น 0 ไม่ขึ้นในกราฟ , ตัวเลขขาวทุกช่อง ช่องเล็กกว่า Min_ ขยายให้พอดีตัวเลข
  VAR C1_ = "#6AA84F"  VAR C2_ = "#E6B000"  VAR C3_ = "#E69138"  VAR C4_ = "#CC0000"
  VAR Min_ = 26
  VAR Dot_ = "<span style='font-size:20px;line-height:14px;vertical-align:-2px;color:"
  // Outlook บนคอมดันชื่อบางแถวลง: แถวละตาราง ไม่ซ้อน , ช่องชื่อกว้างเท่ากันทุกแถวทุกขั้น , ทุกช่องล็อกสูง 22px line-height แบบ exactly
  VAR LW_ = MAX(110, ROUND(MAXX(FILTER(RowsY_, [@n] > 0), LEN([@v]) + LEN(FORMAT([@n], "#,0")) + 3) * 7.5, 0) + 12)
  VAR Legend_ =
    "<p style='margin:4px 0 0;" & F_ & "color:#555'>"   // พี่ปอนด์ 08/11/2026 อยู่ใต้กราฟของแต่ละขั้น ไม่อยู่บนสุดของเมล
      & Dot_ & C1_ & "'>&#9679;</span> &lt;=7 Day&nbsp;&nbsp;&nbsp; "
      & Dot_ & C2_ & "'>&#9679;</span> &lt;=15 Day&nbsp;&nbsp;&nbsp; "
      & Dot_ & C3_ & "'>&#9679;</span> &lt;=30 Day&nbsp;&nbsp;&nbsp; "
      & Dot_ & C4_ & "'>&#9679;</span> &gt;30 Day</p>"
  VAR H_ = "<div style='border-top:1px solid #E0E0E0;padding-top:14px;margin-top:18px'><p style='margin:0 0 10px;" & F_ & "font-size:13px;color:#222'>"   // พี่ปอนด์ 04/11/2026 หัวข้อส่วนไม่ต้องหนา , 09/11/2026 เปลี่ยนเป็นหัวขั้นตัวหนา (ใส่ที่ span ของชื่อขั้นกับยอด)
  // ก้อนละขั้น: หัวขั้น (ยอด) , กราฟสามปีแถวเดียว , ตารางทีละปี (พี่ปอนด์ 08/11/2026) สเกลแท่งเดียวทั้งเมล (ฟ้าลองสเกลแยกขั้นแล้ว งาน 1 งานในขั้นเล็กวาดยาวเท่า 56 งานในขั้นอื่น อ่านผิด)
  VAR MaxN_ = MAX(1, MAXX(RowsY_, [@n]))
  // ป้าย <!-- stage --> <!-- chart --> <!-- table --> <!-- year --> ให้ check_v2.py แยกอ่านจากเนื้อเมลที่ส่งจริง
  VAR Blocks_ =
    CONCATENATEX(StageInfo_,
      VAR s = [@s]
      VAR TblOnly = IF(TableOnlyAll_, TRUE(), IF(Layout6_, NOT [@chart6], TableOnlySmall_ && [@small]))   // ขั้นนี้มีแต่ตาราง
      VAR RS = FILTER(RowsY_, [@ss] = s)
      VAR ns = SUMX(RS, [@n])
      VAR k = DIVIDE(Bars_, MaxN_)
      VAR GH = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE([@grp], "&", "&amp;"), "<", "&lt;"), ">", "&gt;")
      VAR SH = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE([@full], "&", "&amp;"), "<", "&lt;"), ">", "&gt;")
      VAR Chart =
        "<table border='0' cellpadding='0' cellspacing='0' style='border-collapse:collapse'><tr>"
          & CONCATENATEX(YearsT_,
              VAR y = [@yy]
              VAR R = FILTER(RS, [@yy] = y)
              VAR ny = SUMX(R, [@n])
              RETURN "<td valign='top' style='padding:0 36px 0 0;vertical-align:top'><!-- year " & y & " -->" & YP_ & "0 0 8px'>ปี " & y & " (" & FORMAT(ny, "#,0") & ")</p>"
                & IF(ny = 0, NoJob_,
                  CONCATENATEX(FILTER(R, [@n] > 0),
                    VAR w1 = MAX(Min_, ROUND([@a1] * k, 0))  VAR w2 = MAX(Min_, ROUND([@a2] * k, 0))  VAR w3 = MAX(Min_, ROUND([@a3] * k, 0))  VAR w4 = MAX(Min_, ROUND([@a4] * k, 0))
                    RETURN "<table border='0' cellpadding='0' cellspacing='0' style='border-collapse:collapse;margin:0 0 8px'><tr>"
                      & "<td width='" & LW_ & "' height='22' valign='middle' nowrap style='padding:0 12px 0 0;width:" & LW_ & "px;height:22px;" & F_ & "font-size:13px;line-height:22px;mso-line-height-rule:exactly;color:#333'>" & [@h] & " (" & FORMAT([@n], "#,0") & ")</td>"
                      & IF([@a1] > 0, "<td width='" & w1 & "' height='22' bgcolor='" & C1_ & "' align='center' valign='middle' style='width:" & w1 & "px;height:22px;background:" & C1_ & ";" & F_ & "font-size:11px;line-height:22px;mso-line-height-rule:exactly;font-weight:bold;color:#fff'>" & FORMAT([@a1], "#,0") & "</td>", "")
                      & IF([@a2] > 0, "<td width='" & w2 & "' height='22' bgcolor='" & C2_ & "' align='center' valign='middle' style='width:" & w2 & "px;height:22px;background:" & C2_ & ";" & F_ & "font-size:11px;line-height:22px;mso-line-height-rule:exactly;font-weight:bold;color:#fff'>" & FORMAT([@a2], "#,0") & "</td>", "")
                      & IF([@a3] > 0, "<td width='" & w3 & "' height='22' bgcolor='" & C3_ & "' align='center' valign='middle' style='width:" & w3 & "px;height:22px;background:" & C3_ & ";" & F_ & "font-size:11px;line-height:22px;mso-line-height-rule:exactly;font-weight:bold;color:#fff'>" & FORMAT([@a3], "#,0") & "</td>", "")
                      & IF([@a4] > 0, "<td width='" & w4 & "' height='22' bgcolor='" & C4_ & "' align='center' valign='middle' style='width:" & w4 & "px;height:22px;background:" & C4_ & ";" & F_ & "font-size:11px;line-height:22px;mso-line-height-rule:exactly;font-weight:bold;color:#fff'>" & FORMAT([@a4], "#,0") & "</td>", "")
                      & "</tr></table>",
                    "", [@n], DESC, [@v], ASC))
                & "</td>",
              "", [@yy], ASC)
          & "</tr></table>"
      VAR Tables =
        CONCATENATEX(YearsT_,
          VAR y = [@yy]
          VAR R = FILTER(RS, [@yy] = y)
          VAR ny = SUMX(R, [@n])
          VAR TopY = MAXX(TOPN(1, R, [@n], DESC, [@v], ASC), [@v])
          VAR RT = FILTER(R, (ShowEmpty_ && NOT TblOnly) || [@n] > 0)   // แถว Vendor ในตารางปีนี้ ตามสวิตช์ ShowEmpty_
          VAR KT = COUNTROWS(RT)
          RETURN "<!-- year " & y & " -->" & YP_ & IF(y = Y0_, "14px", "18px") & " 0 6px'>ปี " & y & " (" & FORMAT(ny, "#,0") & ")</p>"
            & IF(ny = 0, NoJob_,
              "<table border='1' bordercolor='#D9D9D9' cellpadding='0' cellspacing='0' style='border-collapse:collapse;border:1px solid #D9D9D9'><tr>"
                & TH_ & "left'>GROUP</th>" & TH_ & "left'>SUB_STAGE</th>" & TH_ & "left'>VENDOR</th>" & TH_ & "right'>&lt;=7 Day</th>" & TH_ & "right'>&lt;=15 Day</th>" & TH_ & "right'>&lt;=30 Day</th>"
                & TH_ & "right'>&gt;30 Day</th>" & TH_ & "right'>REJECT</th></tr>"
                & CONCATENATEX(RT,
                    "<tr>" & IF([@v] = TopY, TD_ & "vertical-align:top' rowspan='" & KT & "'>" & GH & "</td>" & TD_ & "vertical-align:top' rowspan='" & KT & "'>" & SH & "</td>", "")
                      & TD_ & "'>" & [@h] & "</td>"
                      & TD_ & "text-align:right'>" & IF([@a1] = 0, Z_, FORMAT([@a1], "#,0")) & "</td>"
                      & TD_ & "text-align:right'>" & IF([@a2] = 0, Z_, FORMAT([@a2], "#,0")) & "</td>"
                      & TD_ & "text-align:right'>" & IF([@a3] = 0, Z_, FORMAT([@a3], "#,0")) & "</td>"
                      & TD_ & "text-align:right'>" & IF([@a4] = 0, Z_, "<span style='color:#C00000'>" & FORMAT([@a4], "#,0") & "</span>") & "</td>"
                      & TD_ & "text-align:right'>" & IF([@n] = 0, Z_, FORMAT([@n], "#,0")) & "</td></tr>",
                    "", [@n], DESC, [@v], ASC)
                & "<tr>" & TF_ & "left' colspan='3'>Grand Total</td>"
                & TF_ & "right'>" & FORMAT(SUMX(R, [@a1]), "#,#") & "</td>" & TF_ & "right'>" & FORMAT(SUMX(R, [@a2]), "#,#") & "</td>"
                & TF_ & "right'>" & FORMAT(SUMX(R, [@a3]), "#,#") & "</td>" & TF_ & "right;color:#C00000'>" & FORMAT(SUMX(R, [@a4]), "#,#") & "</td>"
                & TF_ & "right'>" & FORMAT(ny, "#,#") & "</td></tr></table>"),
          "", [@yy], ASC)
      // แบบ 4 (พี่ปอนด์ 08/11/2026): ขั้นงานน้อยรวมสามปีเป็นตารางเดียว คอลัมน์ ปี (ยอด) รวมช่องต่อปี เฉพาะแถวที่มีงาน
      VAR RSn = FILTER(RS, [@n] > 0)
      // แบบ 5 (พี่ปอนด์ 08/11/2026): ปีที่ไม่มีงานยังขึ้นเป็นแถว ไม่มีงาน Reject (สวิตช์ ShowEmptyYearSmall_)
      VAR EmptyY = IF(ShowEmptyYearSmall_, COUNTROWS(FILTER(YearsT_, VAR yy = [@yy] RETURN COUNTROWS(FILTER(RSn, [@yy] = yy)) = 0)), 0)
      // แบบ 7 (พี่ปอนด์ 09/11/2026): แถว Sub Total ต่อปี เฉพาะปีที่มี Vendor ตั้งแต่ 2 ราย นับเป็นแถวของ rowspan GROUP ด้วย
      VAR SubY = IF(SubTotalYear_, COUNTROWS(FILTER(YearsT_, VAR yy = [@yy] RETURN COUNTROWS(FILTER(RSn, [@yy] = yy)) >= 2)), 0)
      VAR NR = COUNTROWS(RSn) + EmptyY + SubY
      VAR Y1 = IF(ShowEmptyYearSmall_, Y0_, MINX(RSn, [@yy]))
      VAR Comb =
        "<!-- combined --><table border='1' bordercolor='#D9D9D9' cellpadding='0' cellspacing='0' style='border-collapse:collapse;border:1px solid #D9D9D9;margin-top:4px'><tr>"
          & TH_ & "left'>GROUP</th>" & TH_ & "left'>SUB_STAGE</th>" & TH_ & "left'>ปี</th>" & TH_ & "left'>VENDOR</th>" & TH_ & "right'>&lt;=7 Day</th>" & TH_ & "right'>&lt;=15 Day</th>" & TH_ & "right'>&lt;=30 Day</th>"
          & TH_ & "right'>&gt;30 Day</th>" & TH_ & "right'>REJECT</th></tr>"
          & CONCATENATEX(YearsT_,
              VAR y = [@yy]
